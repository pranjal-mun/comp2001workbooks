1. Library classes and packages

A library class supplies behaviour we can reuse. We need its public operations and meaning, without needing its method bodies. A package groups related classes. A qualified name combines the package and class; an import makes a short name available and copies no source code. Java imports java.lang automatically.

2. Documentation and generic types

Public class documentation describes its purpose, constructors, method headers, and behaviour. Detailed descriptions explain argument restrictions. The interface is what callers can use; the implementation is the code and private state behind it. For example, substring(int beginIndex, int endIndex) includes its beginning and excludes its end, a detail its name alone does not tell us.

7. Writing documentation and hiding information

A Javadoc comment begins with /** immediately before a declaration. A class comment describes purpose and may use @author and @version. A method comment describes behaviour, its parameters with @param, and its result with @return. Omit the return tag for void methods and constructors. Private helpers also need useful comments for source readers.

ClinicSchedule.java 
import java.util.ArrayList;

public class ClinicSchedule
{
    private ArrayList<Appointment> appointments;

    public ClinicSchedule()
    {
        appointments = new ArrayList<>();
    }

    public void addAppointment(Appointment appointment)
    {
        appointments.add(appointment);
    }

    public int getCount()
    {
        return appointments.size();
    }

    public void printPatientAt(int index)
    {
        if(index < 0 || index >= appointments.size()) {
            System.out.println("Invalid index");
            return;
        }

        System.out.println(appointments.get(index).getPatientName());
    }

    public boolean cancel(int index)
    {
        if(index < 0 || index >= appointments.size()) {
            return false;
        }

        if(appointments.get(index).hasCheckedIn()) {
            return false;
        }

        appointments.remove(index);
        return true;
    }

    public void printWaitingPatients()
    {
        boolean found = false;

        for(Appointment appointment : appointments) {
            if(!appointment.hasCheckedIn()) {
                System.out.println(appointment.getPatientName());
                found = true;
            }
        }

        if(!found) {
            System.out.println("No waiting patients");
        }
    }

    public int findPatientIndex(String name)
    {
        int index = 0;
        boolean found = false;

        while(index < appointments.size() && !found) {
            Appointment appointment = appointments.get(index);
            if(appointment.getPatientName().equals(name)) {
                found = true;
            }
            else {
                index++;
            }
        }

        if(found) {
            return index;
        }
        return -1;
    }

    public void printAllPatients()
    {
        appointments.forEach(appointment -> System.out.println(appointment.getPatientName()));
    }

    public void checkInAll()
    {
        appointments.forEach(Appointment::checkIn);
    }

    public int removeCheckedIn()
    {
        int before = appointments.size();
        appointments.removeIf(appointment -> appointment.hasCheckedIn());
        return before - appointments.size();
    }

    public boolean cancelPatient(String name)
    {
        return appointments.removeIf(appointment -> appointment.getPatientName().equals(name)
            && !appointment.hasCheckedIn());
    }

    public long countWaiting()
    {
        return appointments.stream()
                           .filter(appointment -> !appointment.hasCheckedIn())
                           .count();
    }

    public void printWaitingNames()
    {
        appointments.stream()
                    .filter(appointment -> !appointment.hasCheckedIn())
                    .map(appointment -> appointment.getPatientName())
                    .forEach(name -> System.out.println(name));
    }

    public int totalWaitingMinutes()
    {
        return appointments.stream()
                           .filter(appointment -> !appointment.hasCheckedIn())
                           .map(appointment -> appointment.getMinutes())
                           .reduce(0, (total, minutes) -> total + minutes);
    }
}

import java.util.ArrayList;
import java.util.Random;

public class ClinicSchedule
{
    private ArrayList<Appointment> appointments;
    private Random randomGenerator;

    public ClinicSchedule()
    {
        appointments = new ArrayList<>();
        randomGenerator = new Random(42);
    }

    public void addAppointment(Appointment appointment)
    {
        appointments.add(appointment);
    }

    public int getCount()
    {
        return appointments.size();
    }

    public void printPatientAt(int index)
    {
        if(index < 0 || index >= appointments.size()) {
            System.out.println("Invalid index");
            return;
        }

        System.out.println(appointments.get(index).getPatientName());
    }

    public boolean cancel(int index)
    {
        if(index < 0 || index >= appointments.size()) {
            return false;
        }

        if(appointments.get(index).hasCheckedIn()) {
            return false;
        }

        appointments.remove(index);
        return true;
    }

    public void printWaitingPatients()
    {
        boolean found = false;

        for(Appointment appointment : appointments) {
            if(!appointment.hasCheckedIn()) {
                System.out.println(appointment.getPatientName());
                found = true;
            }
        }

        if(!found) {
            System.out.println("No waiting patients");
        }
    }

    public int findPatientIndex(String name)
    {
        int index = 0;
        boolean found = false;

        while(index < appointments.size() && !found) {
            Appointment appointment = appointments.get(index);
            if(appointment.getPatientName().equals(name)) {
                found = true;
            }
            else {
                index++;
            }
        }

        if(found) {
            return index;
        }
        return -1;
    }

    public void printAllPatients()
    {
        appointments.forEach(appointment -> System.out.println(appointment.getPatientName()));
    }

    public void checkInAll()
    {
        appointments.forEach(Appointment::checkIn);
    }

    public int removeCheckedIn()
    {
        int before = appointments.size();
        appointments.removeIf(appointment -> appointment.hasCheckedIn());
        return before - appointments.size();
    }

    public boolean cancelPatient(String name)
    {
        return appointments.removeIf(appointment -> appointment.getPatientName().equals(name)
            && !appointment.hasCheckedIn());
    }

    public long countWaiting()
    {
        return appointments.stream()
                           .filter(appointment -> !appointment.hasCheckedIn())
                           .count();
    }

    public void printWaitingNames()
    {
        appointments.stream()
                    .filter(appointment -> !appointment.hasCheckedIn())
                    .map(appointment -> appointment.getPatientName())
                    .forEach(name -> System.out.println(name));
    }

    public int totalWaitingMinutes()
    {
        return appointments.stream()
                           .filter(appointment -> !appointment.hasCheckedIn())
                           .map(appointment -> appointment.getMinutes())
                           .reduce(0, (total, minutes) -> total + minutes);
    }

    /**
     * Find the first patient whose name matches after cleaning both names.
     * Leading and trailing spaces and differences in case are ignored.
     * @param name The name to search for; must not be null.
     * @return The first matching index, or -1 if no name matches.
     */
    public int findPatientIndexIgnoringCase(String name)
    {
        String cleanedName = name.trim().toLowerCase();
        int index = 0;

        while(index < appointments.size()) {
            String storedName = appointments.get(index).getPatientName();
            if(storedName.trim().toLowerCase().equals(cleanedName)) {
                return index;
            }
            index++;
        }
        return -1;
    }

    /**
     * Choose a patient name from all current appointments at random.
     * Checked-in appointments are included.
     * @return A stored patient name, or null if the schedule is empty.
     */
    public String randomPatientName()
    {
        if(appointments.isEmpty()) {
            return null;
        }
        int index = randomGenerator.nextInt(appointments.size());
        return appointments.get(index).getPatientName();
    }
}
